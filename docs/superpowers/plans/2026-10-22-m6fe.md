## 設計決定

1. **provider 選擇順序與平台參數**：`loadConfig(env, platform = process.platform)` 多一個 `platform` 參數，讓測試能在任何平台驗證「Windows 才會退回 `windows`」。沒有任何可用 provider 時，錯誤訊息仍以既有的 `No transcription API key: set GROQ_API_KEY (free key at https://console.groq.com/keys) or OPENAI_API_KEY` 開頭（現有測試與 README 引用這段），後面再補一句 Windows 說明。
2. **`TranscriberConfig` 改成判別聯集**：`windows` 不需要金鑰，所以是 `{ provider: 'windows'; model: 'System.Speech zh-TW'; ffmpegPath: string }`；雲端維持 `{ provider: 'groq' | 'openai'; apiKey; model }`。`model` 兩邊都有，`server.ts` 的啟動紀錄不用改格式。
3. **逾時 30 秒從開始轉檔算起**：照設計 1.2a 的 30 秒，涵蓋 ffmpeg 與 PowerShell 兩步（共用一個期限），排隊時間不算。實測 12 秒的段落辨識約 1 秒，30 秒只是保險；前端 30 秒逾時後會重送，用戶端斷線時 `signal` 會把正在跑的子程序 kill 掉。
4. **ffmpeg 檢查在 `server.ts`**：`loadConfig` 保持純函式（不執行外部程式）；`server.ts` 在 provider 為 `windows` 時用 `findFfmpegProblem` 執行一次 `ffmpeg -version`，失敗就印一行 `[config] ffmpeg not found ...` 並 exit 1。
5. **TTS 腳本共用**：`apps/api/scripts/windows-tts.ps1` 同時給後端整合測試與網頁真實語音 e2e 使用，輸出 48 kHz、16-bit、單聲道 WAV（Chromium 假麥克風與 ffmpeg 都直接吃）。兩支 `.ps1` 都存成 UTF-8 with BOM，並有測試把關。
6. **`#oc-setup` 一律移除**：只要 hash 裡有 `oc-setup`，不論內容合不合法都用 `history.replaceState` 移除（內容可能含存取金鑰，不該留在網址列或歷史紀錄）；不合法時不寫入、不報錯，保留原設定。
7. **原型啟動器的測試放進 `test:workflow`**：`scripts/prototype/*.test.mjs` 用 `node:test`。根 `test:workflow` 改成同時跑 `scripts/ci/*.test.mjs` 與 `scripts/prototype/*.test.mjs`，CI 已經在跑 `npm run test:workflow`，不必修改 `ci.yml`。
8. **啟動器讀三層環境變數**：`process.env` > `apps/api/.env.local`（啟動器寫的金鑰）> `apps/api/.env`（使用者自己的金鑰）；空白值不會蓋掉下一層。後端子程序直接拿合併後的環境，不再依賴 `--env-file`，所以啟動器判斷的 provider 和後端實際選的一致。
9. **真實語音 e2e 不讀 `apps/api/.env`**：後端以 `node --import tsx src/server.ts` 啟動，provider 只看這個 shell 的環境變數，測試才能準確回報這次用的 provider（也印在 `stdout: 'pipe'` 的後端啟動紀錄裡）。測試音訊靠 Windows TTS 合成，所以非 Windows 一律 skip（設計 1.4 的「非 Windows 而且沒有金鑰才 skip」在非 Windows 上無法合成音訊，只能整個 skip）。
10. **既有錄音測試**：`segmentRecorder.test.ts` 大多數測試在驗證錄音器本身（切段、停止、中斷），不是預設值，所以在 `setup()` 固定成 4–8 秒，另加一個測試確認預設是 12 秒強制切段。網頁錄音 e2e 的等待上限從 15 秒放寬到 25 秒。

```ts
/** RMS below this counts as silence unless a caller passes its own threshold. */
export const DEFAULT_SILENCE_THRESHOLD = 0.01;

/**
 * Default segment length limits. Groq bills every request as at least 10 s of audio, so 8–12 s
 * segments keep an hour of recording near 6,000 billed seconds, under the free tier's 7,200 per
 * hour (add-ons design 1.2).
 */
export const DEFAULT_MIN_SEGMENT_MS = 8000;
export const DEFAULT_MAX_SEGMENT_MS = 12000;

export interface SegmentPolicyOptions {
  /** No cut before this much audio (default 8000ms). */
  minMs?: number;
  /** Always cut at this length (default 12000ms). */
  maxMs?: number;
  /** A pause must last this long (default 300ms). */
  silenceMs?: number;
  /** RMS below this counts as silence (default 0.01). */
  threshold?: number;
}

/**
 * Decides when to end a recording segment: at a pause (RMS < threshold for silenceMs) once minMs
 * have passed, or at maxMs regardless. Call start() at the beginning of every segment.
 */
export class SegmentPolicy {
  private readonly minMs: number;
  private readonly maxMs: number;
  private readonly silenceMs: number;
  private readonly threshold: number;
  private startMs = 0;
  private silenceStartMs: number | null = null;

  constructor(options: SegmentPolicyOptions = {}) {
    this.minMs = options.minMs ?? DEFAULT_MIN_SEGMENT_MS;
    this.maxMs = options.maxMs ?? DEFAULT_MAX_SEGMENT_MS;
    this.silenceMs = options.silenceMs ?? 300;
    this.threshold = options.threshold ?? DEFAULT_SILENCE_THRESHOLD;
    for (const [name, value] of Object.entries({
      minMs: this.minMs,
      maxMs: this.maxMs,
      silenceMs: this.silenceMs,
      threshold: this.threshold,
    })) {
      if (!Number.isFinite(value) || value < 0) {
        throw new RangeError(`SegmentPolicy ${name} must be a finite number >= 0, got ${value}`);
      }
    }
    if (this.minMs > this.maxMs) {
      throw new RangeError(`SegmentPolicy minMs (${this.minMs}) must not exceed maxMs (${this.maxMs})`);
    }
  }
```

```ts
import { spawn } from 'node:child_process';

/** stderr 只保留這麼多字給錯誤訊息用，避免囉嗦的工具吃掉記憶體。 */
export const MAX_STDERR_CHARS = 4000;

export interface ProcessResult {
  /** 結束碼；因逾時或 abort 被 kill 時為 null。 */
  code: number | null;
  stdout: Buffer;
  stderr: string;
  timedOut: boolean;
}

export interface RunOptions {
  timeoutMs: number;
  signal?: AbortSignal;
}

export type ProcessRunner = (
  command: string,
  args: readonly string[],
  options: RunOptions,
) => Promise<ProcessResult>;

/**
 * 不經 shell 執行外部程式（參數原樣傳入，不會被 cmd.exe 解析），收集輸出。
 * 超過 timeoutMs 或 signal abort 時 kill 掉程序，promise 以 code null 結束；
 * 只有程式根本無法啟動（例如 ENOENT）時才 reject。
 */
export const runProcess: ProcessRunner = (command, args, { timeoutMs, signal }) =>
  new Promise((resolve, reject) => {
    if (signal?.aborted) {
      resolve({ code: null, stdout: Buffer.alloc(0), stderr: '', timedOut: false });
      return;
    }
    const child = spawn(command, [...args], { stdio: ['ignore', 'pipe', 'pipe'], windowsHide: true });
    const chunks: Buffer[] = [];
    let stderr = '';
    let timedOut = false;
    const kill = () => {
      if (child.exitCode === null && child.signalCode === null) child.kill();
    };
    const timer = setTimeout(() => {
      timedOut = true;
      kill();
    }, timeoutMs);
    signal?.addEventListener('abort', kill, { once: true });
    const cleanup = () => {
      clearTimeout(timer);
      signal?.removeEventListener('abort', kill);
    };
    child.stdout.on('data', (chunk: Buffer) => chunks.push(chunk));
    child.stderr.setEncoding('utf8');
    child.stderr.on('data', (chunk: string) => {
      if (stderr.length < MAX_STDERR_CHARS) stderr = (stderr + chunk).slice(0, MAX_STDERR_CHARS);
    });
    child.once('error', (err) => {
      cleanup();
      reject(err);
    });
    child.once('close', (code) => {
      cleanup();
      resolve({ code, stdout: Buffer.concat(chunks), stderr, timedOut });
    });
  });

export type Limiter = <T>(task: () => Promise<T>) => Promise<T>;

/** 最多同時執行 maxConcurrent 個 task，其餘依呼叫順序排隊。 */
export function createLimiter(maxConcurrent: number): Limiter {
  if (!Number.isInteger(maxConcurrent) || maxConcurrent < 1) {
    throw new RangeError(`maxConcurrent must be an integer >= 1, got ${maxConcurrent}`);
  }
  let active = 0;
  const waiting: (() => void)[] = [];
  const acquire = (): Promise<void> => {
    if (active < maxConcurrent) {
      active += 1;
      return Promise.resolve();
    }
    return new Promise((resolve) => waiting.push(resolve));
  };
  // 有人排隊時直接把名額交給下一個，active 不變。
  const release = () => {
    const next = waiting.shift();
    if (next) next();
    else active -= 1;
  };
  return async <T>(task: () => Promise<T>): Promise<T> => {
    await acquire();
    try {
      return await task();
    } finally {
      release();
    }
  };
}
```

import { spawnSync } from 'node:child_process';
import { mkdtemp, rm, writeFile } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { extname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { createLimiter, runProcess, type ProcessResult, type ProcessRunner } from './processRunner';
import type { Transcriber } from './types';

/** 啟動紀錄顯示的「模型」名稱。 */
export const WINDOWS_STT_MODEL = 'System.Speech zh-TW';
/** 一段音訊從轉檔到辨識完成的期限（ffmpeg 與 PowerShell 共用），不含排隊時間。 */
export const WINDOWS_STT_TIMEOUT_MS = 30_000;
/** 同時處理的段數上限，其餘排隊。 */
export const WINDOWS_STT_CONCURRENCY = 2;
/** FFMPEG_PATH 沒設定時，用 PATH 上的 ffmpeg。 */
export const DEFAULT_FFMPEG_PATH = 'ffmpeg';
/** UTF-8 with BOM 的辨識腳本（apps/api/scripts/windows-stt.ps1）。 */
export const WINDOWS_STT_SCRIPT = fileURLToPath(
  new URL('../../scripts/windows-stt.ps1', import.meta.url),
);

/** ffmpeg 或辨識腳本失敗、逾時、輸出格式不對；app.ts 對應成 502 upstream_error。 */
export class WindowsTranscribeError extends Error {
  constructor(message: string) {
    super(message);
    this.name = 'WindowsTranscribeError';
  }
}

export interface WindowsTranscriberOptions {
  ffmpegPath?: string;
  /** 只給測試注入假的外部程式。 */
  run?: ProcessRunner;
  timeoutMs?: number;
  concurrency?: number;
  scriptPath?: string;
  /** 暫存目錄的上層，預設 os.tmpdir()。 */
  tempDir?: string;
}

/** 暫存檔沿用上傳檔的副檔名（ffmpeg 會再自己判斷格式）；看起來怪的一律用 .bin。 */
function inputExtension(filename: string): string {
  const ext = extname(filename).toLowerCase();
  return /^\.[a-z0-9]{1,8}$/.test(ext) ? ext : '.bin';
}

function assertSucceeded(result: ProcessResult, step: string, timeoutMs: number): void {
  if (result.timedOut) throw new WindowsTranscribeError(`${step} timed out after ${timeoutMs}ms`);
  if (result.code !== 0) {
    const detail = result.stderr
      .trim()
      .split(/\r?\n/)
      .filter((line) => line.trim() !== '')
      .slice(-3)
      .join(' | ')
      .slice(0, 300);
    throw new WindowsTranscribeError(
      `${step} exited with code ${result.code}${detail ? `: ${detail}` : ''}`,
    );
  }
}

/** 解析 windows-stt.ps1 的輸出 `{"text":"..."}`。 */
export function parseSttOutput(stdout: Buffer): string {
  let parsed: unknown;
  try {
    parsed = JSON.parse(stdout.toString('utf8').replace(/^\uFEFF/, '').trim());
  } catch {
    throw new WindowsTranscribeError('windows-stt printed invalid JSON');
  }
  if (
    typeof parsed !== 'object' ||
    parsed === null ||
    typeof (parsed as { text?: unknown }).text !== 'string'
  ) {
    throw new WindowsTranscribeError('windows-stt output has no "text" string');
  }
  return (parsed as { text: string }).text;
}

/**
 * Windows 內建離線轉寫（add-ons 設計 1.2a）：音訊寫進暫存檔 → ffmpeg 轉成 16 kHz 單聲道 WAV →
 * PowerShell 用 System.Speech（zh-TW 聽寫）辨識。忽略 prompt；輸出的繁體轉換由 app.ts 統一處理。
 */
export function createWindowsTranscriber(options: WindowsTranscriberOptions = {}): Transcriber {
  const {
    ffmpegPath = DEFAULT_FFMPEG_PATH,
    run = runProcess,
    timeoutMs = WINDOWS_STT_TIMEOUT_MS,
    concurrency = WINDOWS_STT_CONCURRENCY,
    scriptPath = WINDOWS_STT_SCRIPT,
    tempDir = tmpdir(),
  } = options;
  const limit = createLimiter(concurrency);

  async function transcribeOnce(file: File, signal: AbortSignal | undefined): Promise<string> {
    // 排隊期間用戶端已斷線：不必再轉檔。
    signal?.throwIfAborted();
    const deadline = Date.now() + timeoutMs;
    const remaining = () => Math.max(1, deadline - Date.now());
    const dir = await mkdtemp(join(tempDir, 'oc-stt-'));
    try {
      const input = join(dir, `input${inputExtension(file.name)}`);
      const wav = join(dir, 'audio.wav');
      await writeFile(input, new Uint8Array(await file.arrayBuffer()));
      const converted = await run(
        ffmpegPath,
        ['-nostdin', '-hide_banner', '-loglevel', 'error', '-y', '-i', input, '-ar', '16000', '-ac', '1', '-f', 'wav', wav],
        { timeoutMs: remaining(), signal },
      );
      assertSucceeded(converted, 'ffmpeg', timeoutMs);
      const recognized = await run(
        'powershell.exe',
        ['-NoProfile', '-NonInteractive', '-ExecutionPolicy', 'Bypass', '-File', scriptPath, '-Wav', wav],
        { timeoutMs: remaining(), signal },
      );
      assertSucceeded(recognized, 'windows-stt', timeoutMs);
      return parseSttOutput(recognized.stdout);
    } finally {
      // 刪不掉（例如被 kill 的程序還沒放開檔案）只記錄，不蓋掉原本的結果或錯誤。
      await rm(dir, { recursive: true, force: true, maxRetries: 3, retryDelay: 100 }).catch(
        (err: unknown) =>
          console.warn(
            '[transcribe] could not delete temp files:',
            err instanceof Error ? err.message : String(err),
          ),
      );
    }
  }

  return {
    transcribe({ file, signal }) {
      return limit(() => transcribeOnce(file, signal));
    },
  };
}

/** 執行一次 `ffmpeg -version`，能跑就回 true。 */
export function probeFfmpeg(ffmpegPath: string): boolean {
  const result = spawnSync(ffmpegPath, ['-version'], {
    stdio: 'ignore',
    windowsHide: true,
    timeout: 10_000,
  });
  return result.error === undefined && result.status === 0;
}

/** server.ts 啟動時檢查：找不到 ffmpeg 就回傳要印出的訊息。 */
export function findFfmpegProblem(
  ffmpegPath: string,
  probe: (path: string) => boolean = probeFfmpeg,
): string | undefined {
  if (probe(ffmpegPath)) return undefined;
  return (
    `ffmpeg not found at "${ffmpegPath}": TRANSCRIBE_PROVIDER=windows needs ffmpeg ` +
    '(install it, e.g. winget install Gyan.FFmpeg, or set FFMPEG_PATH to ffmpeg.exe)'
  );
}

/**
 * loadConfig 產生的轉寫服務設定；model 一律是實際使用的模型名稱（windows 為 'System.Speech zh-TW'）。
 * config.ts 應從這裡匯入 TranscriberConfig／TranscribeProvider，不要重複宣告。
 */
export type TranscriberConfig =
  | { provider: 'groq' | 'openai'; apiKey: string; model: string }
  | { provider: 'windows'; model: string; ffmpegPath: string };

/** 依設定建出轉寫服務。options.fetch 只給測試注入假上游，server.ts 不傳。 */
export function createTranscriber(
  config: TranscriberConfig,
  options: UpstreamClientOptions = {},
): Transcriber {
  switch (config.provider) {
    case 'groq':
      return createGroqTranscriber(
        config.apiKey,
        config.model,
        createGroqClient(config.apiKey, options),
      );
    case 'openai':
      // OpenAI 固定用 TRANSCRIBE_MODEL；config.model 只供啟動訊息顯示，不同就擋下，免得覆寫被默默忽略。
      if (config.model !== TRANSCRIBE_MODEL) {
        throw new Error(
          `OpenAI transcription always uses ${TRANSCRIBE_MODEL}; got model "${config.model}"`,
        );
      }
      return createOpenAITranscriber(config.apiKey, createOpenAIClient(config.apiKey, options));
    case 'windows':
      // 離線辨識不連網，options.fetch 用不到。
      return createWindowsTranscriber({ ffmpegPath: config.ffmpegPath });
    default: {
      const unknownProvider: never = config;
      throw new Error(`Unknown transcribe provider: ${String((unknownProvider as { provider: unknown }).provider)}`);
    }
  }
}

function loadTranscriberConfig(
  env: Record<string, string | undefined>,
  platform: NodeJS.Platform,
): TranscriberConfig {
  const groqKey = readValue(env, 'GROQ_API_KEY');
  const openaiKey = readValue(env, 'OPENAI_API_KEY');
  const rawProvider = readValue(env, 'TRANSCRIBE_PROVIDER');

  let provider: TranscribeProvider;
  if (rawProvider === undefined) {
    if (groqKey) provider = 'groq';
    else if (openaiKey) provider = 'openai';
    else if (platform === 'win32') provider = 'windows';
    else throw new Error(NO_PROVIDER_MESSAGE);
  } else {
    const normalized = rawProvider.toLowerCase();
    if (normalized !== 'groq' && normalized !== 'openai' && normalized !== 'windows') {
      // 若誤把金鑰貼進 TRANSCRIBE_PROVIDER，不要讓它出現在啟動日誌裡。

      const shown = looksLikeKey ? 'an unrecognized value (not shown)' : `"${rawProvider}"`;
      throw new Error(`TRANSCRIBE_PROVIDER must be "groq", "openai" or "windows", got ${shown}`);
    }
    provider = normalized;
  }

  if (provider === 'windows') {
    if (platform !== 'win32') {
      throw new Error(
        'TRANSCRIBE_PROVIDER=windows only works on Windows (it uses System.Speech); set GROQ_API_KEY or OPENAI_API_KEY instead',
      );
    }
    return {
      provider,
      model: WINDOWS_STT_MODEL,
      ffmpegPath: readValue(env, 'FFMPEG_PATH') ?? DEFAULT_FFMPEG_PATH,
    };
  }

  if (provider === 'groq') {
    if (!groqKey) throw new Error('GROQ_API_KEY is required when TRANSCRIBE_PROVIDER=groq');
    return {
      provider,
      apiKey: groqKey,
      model: readValue(env, 'GROQ_TRANSCRIBE_MODEL') ?? GROQ_DEFAULT_TRANSCRIBE_MODEL,
    };
  }

  if (!openaiKey) throw new Error('OPENAI_API_KEY is required when TRANSCRIBE_PROVIDER=openai');
  return { provider, apiKey: openaiKey, model: TRANSCRIBE_MODEL };
}

import { isValidBaseUrl, normalizeSettings, saveSettings, type Settings } from './settingsStore';

/** `#oc-setup=<base64url(JSON)>`，由 `npm run prototype`（之後也包含 M9 的分享網址）產生。 */
export const SETUP_HASH_PARAM = 'oc-setup';
/** 存取金鑰長度上限；正常的金鑰是 16–64 字。 */
export const MAX_SETUP_KEY_LENGTH = 512;
/** 編碼後的長度上限，擋掉離譜的網址。 */
export const MAX_SETUP_PAYLOAD_LENGTH = 4096;

export type SetupLinkResult = 'none' | 'applied' | 'invalid';

/** `{ apiBaseUrl, accessKey }` → 無 padding 的 base64url（與 Node 的 Buffer `base64url` 相同）。 */
export function encodeSetupPayload(settings: Settings): string {
  const json = JSON.stringify({ apiBaseUrl: settings.apiBaseUrl, accessKey: settings.accessKey });
  let binary = '';
  for (const byte of new TextEncoder().encode(json)) binary += String.fromCharCode(byte);
  return btoa(binary).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '');
}

/** 解出並驗證設定；任何不合法（格式、網址、金鑰）都回 null，不丟錯。 */
export function decodeSetupPayload(encoded: string): Settings | null {
  if (encoded.length > MAX_SETUP_PAYLOAD_LENGTH || !/^[A-Za-z0-9_-]+$/.test(encoded)) return null;
  try {
    const base64 = encoded.replace(/-/g, '+').replace(/_/g, '/');
    const binary = atob(base64 + '='.repeat((4 - (base64.length % 4)) % 4));
    const bytes = Uint8Array.from(binary, (char) => char.charCodeAt(0));
    const parsed: unknown = JSON.parse(new TextDecoder('utf-8', { fatal: true }).decode(bytes));
    if (typeof parsed !== 'object' || parsed === null || Array.isArray(parsed)) return null;
    const { apiBaseUrl, accessKey } = parsed as Record<string, unknown>;
    if (typeof apiBaseUrl !== 'string' || typeof accessKey !== 'string') return null;
    const settings = normalizeSettings({ apiBaseUrl, accessKey });
    if (!isValidBaseUrl(settings.apiBaseUrl)) return null;
    if (
      settings.accessKey === '' ||
      /\s/.test(settings.accessKey) ||
      settings.accessKey.length > MAX_SETUP_KEY_LENGTH
    ) {
      return null;
    }
    return settings;
  } catch {
    return null;
  }
}

/** 網址 hash 裡 `oc-setup` 的值；沒有就回 null。 */
function readSetupParam(hash: string): string | null {
  if (!hash.startsWith('#')) return null;
  return new URLSearchParams(hash.slice(1)).get(SETUP_HASH_PARAM);
}

/**
 * App 啟動時呼叫（在讀取設定之前）：hash 有 `oc-setup` 就驗證，合法才存進設定；
 * 不論合不合法都立刻用 history.replaceState 把 hash 移除，因為裡面可能有存取金鑰。
 */
export function applySetupLink(
  win: Pick<Window, 'location' | 'history'> = window,
): SetupLinkResult {
  const encoded = readSetupParam(win.location.hash);
  if (encoded === null) return 'none';
  const settings = decodeSetupPayload(encoded);
  if (settings) saveSettings(settings);
  win.history.replaceState(win.history.state, '', win.location.pathname + win.location.search);
  return settings ? 'applied' : 'invalid';
}

/**
 * Which transcription service the backend will pick, in the order of loadConfig in
 * apps/api/src/config.ts: an explicit TRANSCRIBE_PROVIDER, then Groq, then OpenAI, then (on Windows
 * only) the built-in offline recognizer. The backend still validates the explicit value.
 * @returns {{ kind: 'cloud', provider: string } | { kind: 'windows' } | { kind: 'none' }}
 */
export function chooseTranscriber(env, platform) {
  const explicit = env.TRANSCRIBE_PROVIDER?.trim().toLowerCase();
  if (explicit === 'windows') return platform === 'win32' ? { kind: 'windows' } : { kind: 'none' };
  if (explicit) return { kind: 'cloud', provider: explicit };
  if (env.GROQ_API_KEY?.trim()) return { kind: 'cloud', provider: 'groq' };
  if (env.OPENAI_API_KEY?.trim()) return { kind: 'cloud', provider: 'openai' };
  if (platform === 'win32') return { kind: 'windows' };
  return { kind: 'none' };
}

/** 32 URL-safe random characters: no whitespace or commas, so ACCESS_KEYS accepts it. */

/**
 * The access key the browser will use. The first ACCESS_KEYS entry (from the environment or
 * apps/api/.env(.local)) wins; otherwise a new random key is saved to envLocalPath and reused.
 * @returns {{ key: string, source: 'env' | 'created' }}
 */
export function ensureAccessKey({ env, envLocalPath, createKey = generateAccessKey }) {
  const existing = (env.ACCESS_KEYS ?? '')
    .split(',')
    .map((key) => key.trim())
    .find((key) => key !== '');
  if (existing) return { key: existing, source: 'env' };

  const key = createKey();
  const line = `ACCESS_KEYS=${key}`;
  const before = existsSync(envLocalPath) ? readFileSync(envLocalPath, 'utf8') : '';
  let text;
  if (/^ACCESS_KEYS=.*$/m.test(before)) {
    // Only a blank ACCESS_KEYS= can be here (a filled one was returned above).
    text = before.replace(/^ACCESS_KEYS=.*$/m, line);
  } else {
    const separator = before === '' || before.endsWith('\n') ? '' : '\n';
    text = `${before}${separator}# Written by npm run prototype; reused on every run.\n${line}\n`;
  }
  writeFileSync(envLocalPath, text, 'utf8');
  return { key, source: 'created' };
}

/** base64url(JSON) without padding, as packages/core/src/settings/setupLink.ts decodes it. */

/** The command that opens `url` in the default browser. The URL is never echoed in errors (it holds a key). */
export function openBrowserCommand(url, platform) {
  if (!/^https?:\/\/[\w.:/#=-]+$/.test(url)) throw new Error('refusing to open an unexpected URL');
  if (platform === 'win32') {
    // Start-Process keeps the #fragment; the URL has no quotes (checked above).
    return {
      command: 'powershell.exe',
      args: ['-NoProfile', '-NonInteractive', '-Command', `Start-Process '${url}'`],
    };
  }
  if (platform === 'darwin') return { command: 'open', args: [url] };
  return { command: 'xdg-open', args: [url] };
}

/** Polls `url` until it answers 2xx (true), the timeout passes or isCancelled() (false). */
export async function waitForOk(
  url,
  {
    timeoutMs = 60_000,
    intervalMs = 500,
    fetchImpl = fetch,
    now = Date.now,
    sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms)),
    isCancelled = () => false,
  } = {},
) {
  const deadline = now() + timeoutMs;
  while (!isCancelled()) {
    try {
      const response = await fetchImpl(url, { signal: AbortSignal.timeout(5_000) });
      if (response.ok) return true;
    } catch {
      // Not listening yet.
    }
    if (now() >= deadline) return false;
    await sleep(intervalMs);
  }
  return false;
}

/**
 * Watches child processes: when the first one exits (or cannot start), the others are killed.
 * `done` resolves once every child has exited, with the name and exit code of the first one.
 */
export function supervise(children) {
  const entries = Object.entries(children);
  const exited = new Set();
  let first;
  let resolveDone;
  const done = new Promise((resolve) => {
    resolveDone = resolve;
  });
  const killed = new Set();
  const stopAll = () => {
    for (const [name, child] of entries) {
      if (exited.has(name) || killed.has(name)) continue;
      killed.add(name);
      child.kill();
    }
  };
  const settle = (name, code) => {
    if (exited.has(name)) return;
    exited.add(name);
    if (!first) {
      first = { name, code };
      stopAll();
    }
    if (exited.size === entries.length) resolveDone(first);
  };
  for (const [name, child] of entries) {
    child.once('exit', (code) => settle(name, code));
    child.once('error', () => settle(name, null));
  }
  return {
    done,
    stopAll,
    get anyExited() {
      return exited.size > 0;
    },
  };
}

/**
 * 後端這次會用的 provider，順序同 apps/api/src/config.ts：TRANSCRIBE_PROVIDER → Groq → OpenAI →
 * Windows 離線辨識。
 */
export function resolveRealProvider(
  env: Record<string, string | undefined>,
  platform: NodeJS.Platform,
): string | null {
  const explicit = env['TRANSCRIBE_PROVIDER']?.trim().toLowerCase();
  if (explicit) return explicit;
  if (env['GROQ_API_KEY']?.trim()) return 'groq';
  if (env['OPENAI_API_KEY']?.trim()) return 'openai';
  return platform === 'win32' ? 'windows' : null;
}

/** 與 scripts/prototype.mjs 相同的 `#oc-setup=<base64url(JSON)>`。 */
export function encodeSetupHash(setup: { apiBaseUrl: string; accessKey: string }): string {
  return `#oc-setup=${Buffer.from(JSON.stringify(setup), 'utf8').toString('base64url')}`;
}

import { execFileSync } from 'node:child_process';
import { mkdirSync, rmSync, writeFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { REAL_E2E_SUPPORTED, SPEECH_TEXT, TTS_SCRIPT } from './realEnv';

/** 合成測試音訊到 playwright.real.config.ts 指定的 OC_REAL_WAV；回傳的函式在測試後刪除它。 */
export default function globalSetup(): (() => void) | undefined {
  const wav = process.env['OC_REAL_WAV'];
  if (!REAL_E2E_SUPPORTED || !wav) return undefined;
  const dir = dirname(wav);
  mkdirSync(dir, { recursive: true });
  const textFile = join(dir, 'speech.txt');
  writeFileSync(textFile, SPEECH_TEXT, 'utf8');
  const voice = execFileSync(
    'powershell.exe',
    ['-NoProfile', '-NonInteractive', '-ExecutionPolicy', 'Bypass', '-File', TTS_SCRIPT, '-Out', wav, '-TextFile', textFile],
    { encoding: 'utf8', windowsHide: true, timeout: 60_000 },
  ).trim();
  console.log(`[e2e:real] synthesized ${wav} with ${voice}`);
  return () => rmSync(dir, { recursive: true, force: true });
}

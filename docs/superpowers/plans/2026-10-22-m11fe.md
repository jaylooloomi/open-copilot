# M11 前端：AI 摘要與待辦事項 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 解鎖 ActionBar 的「AI 摘要」與「待辦事項」：呼叫已完成的 `POST /ai/summary`、`POST /ai/todos`，在側邊面板顯示結果，可把摘要插入筆記開頭、把待辦事項以 Tiptap TaskList 插入筆記結尾；三種匯出（Markdown、純文字、Word）與 PDF 列印都支援待辦清單；第一次使用 AI 時顯示一次隱私說明。

**Architecture:** 編輯器加入 `@tiptap/extension-list` 的 `TaskList`、`TaskItem`（`nested: true`），匯出器各自處理 `taskList` 節點。`ApiClient` 新增 `summarize()`、`todos()`（JSON body、獨立且比後端 LLM 逾時更長的期限）。新的 `packages/core/src/ai/` 模組放純函式（錯誤訊息、停用原因、送出的文字、插入編輯器）、隱私說明的 localStorage 記錄，以及管理面板狀態的 `useAiAssistant` hook。UI 由 `ActionBar`（解鎖兩顆按鈕）、`AiPrivacyNotice`（對話框）、`AiPanel`（側邊面板）組成，`NotesApp` 負責串接。後端與 `@oc/shared` 的 AI 型別已在 M11-backend 合併，本計畫不改它們的契約。

**Tech Stack:** React 19、Tiptap v3（`@tiptap/extension-list@^3.32.0`，設計文件第 6 節已核准）、lucide-react、docx、Vitest + Testing Library（jsdom）、Playwright（chromium，mock 後端）。

/**
 * Undo/redo keys always count as handled. ProseMirror's undo/redo commands return false when there
 * is nothing to undo or redo, and the browser then runs its own contenteditable undo, which edits
 * the document behind ProseMirror's history (e.g. Ctrl+Shift+Z deleting the last typed words).

const HistoryKeys = Extension.create({
  name: 'historyKeys',
  addKeyboardShortcuts() {
    const undo = () => {
      this.editor.commands.undo();
      return true;
    };
    const redo = () => {
      this.editor.commands.redo();
      return true;
    };
    return { 'Mod-z': undo, 'Mod-Z': undo, 'Mod-Shift-z': redo, 'Mod-Shift-Z': redo, 'Mod-y': redo, 'Mod-Y': redo };
  },
});

    StarterKit.configure({
      heading: { levels: [1, 2, 3] },

    Color,

    TaskList,
    TaskItem.configure({ nested: true }),

並把 `formatListItem` 換成：

```ts
/**
 * Puts a list marker in front of a list item's text and indents its following lines, so nested
 * lists and wrapped lines stay inside the item. The indent is the marker's width unless given:
 * Markdown task items (`- [ ] `) indent by 2, because `[ ]` belongs to the item text.
 */
export function formatListItem(marker: string, body: string, indentWidth: number = marker.length): string {
  const indent = ' '.repeat(indentWidth);
  return body
    .split('\n')
    .map((line, index) => {
      if (index === 0) return `${marker}${line}`.trimEnd();
      return line === '' ? '' : `${indent}${line}`;
    })
    .join('\n');
}
```

```ts
function renderBlock(node: JSONContent): string {
  switch (node.type) {
    case 'paragraph':
      return finishLines(renderInline(node.content));
    case 'heading': {
      // ATX headings are one line, so hard breaks become spaces.
      const text = escapeHeadingEnd(
        finishLines(renderInline(node.content)).replaceAll(HARD_BREAK, ' '),
      );
      const level = Math.min(Math.max(Number(node.attrs?.level) || 1, 1), 5);
      return text ? `${'#'.repeat(level + 1)} ${text}` : '';
    }
    case 'bulletList':
    case 'orderedList':
    case TASK_LIST_TYPE:
      return renderList(node);
    default:
      // Unknown block: its text content, one paragraph per line.
      return nodeText(node)
        .split('\n')
        .map((line) => finishLines(escapeInline(line)))
        .filter((line) => line !== '')
        .join('\n\n');
  }
}

function renderList(list: JSONContent): string {
  const ordered = list.type === 'orderedList';
  const task = list.type === TASK_LIST_TYPE;
  const start = listStart(list);
  return (list.content ?? [])
    .map((item, index) => {
      let marker = '- ';
      if (task) marker = isTaskChecked(item) ? '- [x] ' : '- [ ] ';
      else if (ordered) marker = `${start + index}. `;
      let body = '';
      for (const child of item.content ?? []) {
        const block = renderBlock(child);
        if (block === '') continue;
        if (body !== '') body += canInterruptParagraph(child, block) ? '\n' : '\n\n';
        body += block;
      }
      // GFM needs whitespace after `]` to read a task marker, so an empty item keeps `- [ ] ` untrimmed
      // (formatListItem would trim it to `- [ ]`, a bullet whose text is `[ ]`).
      if (task && body === '') return marker;
      // GFM reads `[ ]` as part of the item text, so nested lines line up after `- `.
      return formatListItem(marker, body, task ? 2 : marker.length);
    })
    .join('\n');
}

function isList(node: JSONContent): boolean {
  return node.type === 'bulletList' || node.type === 'orderedList' || node.type === TASK_LIST_TYPE;
}
```

```ts
import type { JSONContent, Note } from '@oc/shared';
import {
  formatListItem,
  hasInlineChildren,
  isTaskChecked,
  linkRuns,
  linkShowsHref,
  listStart,
  nodeText,
  TASK_CHECKED,
  TASK_LIST_TYPE,
  TASK_UNCHECKED,
} from './common';

/**
 * Plain text: the title, a blank line, then one line per paragraph or list item.
 * Lists keep their `- ` / `1. ` markers, task items start with ☐ / ☑, and links become
 * `text (href)`; every other kind of formatting is dropped.
 */
export function toTxt(note: Note): string {
  const title = note.title.trim();
  const body = (note.content.content ?? [])
    .map(renderBlock)
    .join('\n')
    // Drop blank lines at both ends (e.g. the empty paragraph Tiptap keeps after a final list).
    .replace(/^\n+|\n+$/g, '');
  return [title, body].filter((part) => part !== '').join('\n\n');
}

function renderBlock(node: JSONContent): string {
  switch (node.type) {
    case 'bulletList':
    case 'orderedList':
    case TASK_LIST_TYPE:
      return renderList(node);
    default:
      // Paragraphs, headings and unknown blocks are their text content, with link addresses kept.
      if (typeof node.text === 'string' || !node.content?.length) return nodeText(node);
      if (hasInlineChildren(node)) return inlineText(node.content);
      return node.content.map(renderBlock).join('\n');
  }
}

function inlineText(nodes: JSONContent[]): string {
  return linkRuns(nodes)
    .map(({ href, nodes: run }) => {
      const text = run.map(nodeText).join('');
      return href && text.trim() !== '' && !linkShowsHref(text, href) ? `${text} (${href})` : text;
    })
    .join('');
}

function renderList(list: JSONContent): string {
  const ordered = list.type === 'orderedList';
  const task = list.type === TASK_LIST_TYPE;
  const start = listStart(list);
  return (list.content ?? [])
    .map((item, index) => {
      let marker = '- ';
      if (task) marker = `${isTaskChecked(item) ? TASK_CHECKED : TASK_UNCHECKED} `;
      else if (ordered) marker = `${start + index}. `;
      const body = (item.content ?? []).map(renderBlock).join('\n');
      return formatListItem(marker, body);
    })
    .join('\n');
}
```

function renderBlock(node: JSONContent, context: Context, format: ParagraphFormat): Paragraph[] {
  switch (node.type) {
    case 'paragraph':
      return [textParagraph(node, format)];
    case 'heading': {
      const level = Math.min(Math.max(Number(node.attrs?.level) || 1, 1), 3) as 1 | 2 | 3;
      return [textParagraph(node, { ...format, heading: HEADINGS[level] })];
    }
    case 'bulletList':
    case 'orderedList':
    case TASK_LIST_TYPE:
      return renderAnyList(node, context, 0);
    case 'blockquote':
      return renderBlockquote(node, context, format);
    default: {
      // Unknown block: its text content as a plain paragraph.
      const text = nodeText(node);
      return text === '' ? [] : [new Paragraph({ ...format, children: plainRuns(text) })];
    }
  }
}

function isListNode(node: JSONContent): boolean {
  return node.type === 'bulletList' || node.type === 'orderedList' || node.type === TASK_LIST_TYPE;
}

/** A list at the given nesting depth: task lists get ☐ / ☑ characters, the others Word numbering. */
function renderAnyList(list: JSONContent, context: Context, depth: number): Paragraph[] {
  return list.type === TASK_LIST_TYPE ? renderTaskList(list, context, depth) : renderList(list, context, depth);
}

function renderList(list: JSONContent, context: Context, depth: number): Paragraph[] {
  const level = Math.min(depth, LEVELS - 1);
  let reference = BULLET_REFERENCE;
  if (list.type === 'orderedList') {
    reference = `oc-ordered-${context.orderedLists.length + 1}`;
    context.orderedLists.push({ reference, start: listStart(list) });
  }
  const paragraphs: Paragraph[] = [];
  for (const item of list.content ?? []) {
    let numbered = false;
    for (const child of item.content ?? []) {
      if (isListNode(child)) {
        paragraphs.push(...renderAnyList(child, context, depth + 1));
        continue;
      }
      // The item's first paragraph carries the bullet/number; later ones only line up with it.
      const format: ParagraphFormat = numbered
        ? { indent: { left: levelIndent(level).left } }
        : { numbering: { reference, level } };
      const rendered = renderBlock(child, context, format);
      if (rendered.length > 0) numbered = true;
      paragraphs.push(...rendered);
    }
  }
  return paragraphs;
}

/**
 * Task items are indented paragraphs whose first line starts with ☐ or ☑ (Word has no checkbox
 * numbering). A task item always starts with a paragraph; should it not, the box gets its own line.
 */
function renderTaskList(list: JSONContent, context: Context, depth: number): Paragraph[] {
  const indent = { left: levelIndent(Math.min(depth, LEVELS - 1)).left };
  const paragraphs: Paragraph[] = [];
  for (const item of list.content ?? []) {
    const box = isTaskChecked(item) ? TASK_CHECKED : TASK_UNCHECKED;
    const children = item.content ?? [];
    if (children[0]?.type !== 'paragraph') {
      paragraphs.push(new Paragraph({ indent, children: [new TextRun(box)] }));
    }
    children.forEach((child, index) => {
      if (isListNode(child)) paragraphs.push(...renderAnyList(child, context, depth + 1));
      // Only a leading paragraph carries the box; any other first child was given its own box line above.
      else if (index === 0 && child.type === 'paragraph') paragraphs.push(textParagraph(child, { indent }, `${box} `));
      else paragraphs.push(...renderBlock(child, context, { indent }));
    });
  }
  return paragraphs;
}

function textParagraph(node: JSONContent, options: IParagraphOptions, prefix?: string): Paragraph {
  const alignment = ALIGNMENTS[node.attrs?.textAlign as keyof typeof ALIGNMENTS];
  const runs = inlineRuns(node.content ?? []);
  return new Paragraph({
    ...options,
    ...(alignment ? { alignment } : {}),
    children: prefix ? [new TextRun(prefix), ...runs] : runs,
  });
}

import {
  AI_UPSTREAM_TIMEOUT_MS,
  type AiRequest,
  type AiSummaryResponse,
  type AiTodosResponse,
  type ApiErrorCode,
  type TranscribeResponse,
} from '@oc/shared';

/** A failed API call. `status` is the HTTP status, or 0 for a network error or timeout. */
export class ApiError extends Error {
  readonly status: number;
  readonly retryAfterMs?: number;
  readonly code?: ApiErrorCode;

  constructor(
    message: string,
    status: number,
    options: { retryAfterMs?: number; code?: ApiErrorCode; cause?: unknown } = {},
  ) {
    super(message, { cause: options.cause });
    this.name = 'ApiError';
    this.status = status;
    this.retryAfterMs = options.retryAfterMs;
    this.code = options.code;
  }
}

export interface TranscribeInput {
  audio: Blob;
  seq: number;
  prompt: string;
  filename: string;
}

export interface ApiClient {
  /** GET /health; true only for 200 `{ ok: true }`. Never throws. */
  health(): Promise<boolean>;
  /** POST /transcribe; rejects with ApiError. */
  transcribe(input: TranscribeInput): Promise<TranscribeResponse>;
  /** POST /ai/summary; rejects with ApiError (503 `ai_unavailable` when the backend has no Groq key). */
  summarize(input: AiRequest): Promise<AiSummaryResponse>;
  /** POST /ai/todos; rejects with ApiError like summarize(). */
  todos(input: AiRequest): Promise<AiTodosResponse>;
}

export interface ApiClientOptions {
  baseUrl: string;
  accessKey: string;
  fetchImpl?: typeof fetch;
  /** Abort a request after this long and report it as a network error (status 0). */
  timeoutMs?: number;
  /** The same, for the AI endpoints, whose LLM call takes much longer than a transcription. */
  aiTimeoutMs?: number;
}

export const REQUEST_TIMEOUT_MS = 30_000;
/**
 * The backend gives the LLM AI_UPSTREAM_TIMEOUT_MS and then answers with its own error; waiting
 * 15 s longer lets that answer arrive instead of a bare network error.
 */
export const AI_REQUEST_TIMEOUT_MS = AI_UPSTREAM_TIMEOUT_MS + 15_000;

// `Record` over the union makes the compiler flag a code added to or removed from @oc/shared.
const API_ERROR_CODE_SET: Record<ApiErrorCode, true> = {
  unauthorized: true,
  payload_too_large: true,
  rate_limited: true,
  bad_request: true,
  upstream_error: true,
  ai_unavailable: true,
};

function isApiErrorCode(value: unknown): value is ApiErrorCode {
  return typeof value === 'string' && Object.hasOwn(API_ERROR_CODE_SET, value);
}

export function createApiClient({
  baseUrl,
  accessKey,
  // Looked up on every call so tests can stub the global fetch.
  fetchImpl = (input, init) => fetch(input, init),
  timeoutMs = REQUEST_TIMEOUT_MS,
  aiTimeoutMs = AI_REQUEST_TIMEOUT_MS,
}: ApiClientOptions): ApiClient {
  const root = baseUrl.trim().replace(/\/+$/, '');

  /** Fetches and reads the JSON body under one deadline, so a stalled body also times out. */
  async function send(
    path: string,
    init: RequestInit,
    deadlineMs: number = timeoutMs,
  ): Promise<{ res: Response; body: unknown }> {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), deadlineMs);
    try {
      let res: Response;
      try {
        res = await fetchImpl(`${root}${path}`, { ...init, signal: controller.signal });
      } catch (cause) {
        throw new ApiError('Network error', 0, { cause });
      }
      const body = await readJson(res);
      if (controller.signal.aborted) throw new ApiError('Network error', 0);
      return { res, body };
    } finally {
      clearTimeout(timer);
    }
  }

  /** POSTs `{ title, text }` as JSON to an AI endpoint; resolves only for a 2xx response. */
  async function postAi(
    path: '/ai/summary' | '/ai/todos',
    { title, text }: AiRequest,
  ): Promise<{ res: Response; body: unknown }> {
    const result = await send(
      path,
      {
        method: 'POST',
        headers: { Authorization: `Bearer ${accessKey}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({ title, text }),
      },
      aiTimeoutMs,
    );
    if (!result.res.ok) throw responseError(result.res, result.body);
    return result;
  }

  return {
    async health() {
      try {
        const { res, body } = await send('/health', { method: 'GET' });
        if (!res.ok) return false;
        return isRecord(body) && body.ok === true;
      } catch {
        return false;
      }
    },

    async transcribe({ audio, seq, prompt, filename }) {
      const form = new FormData();
      form.append('audio', audio, filename);
      form.append('seq', String(seq));
      form.append('prompt', prompt);
      const { res, body } = await send('/transcribe', {
        method: 'POST',
        headers: { Authorization: `Bearer ${accessKey}` },
        body: form,
      });
      if (!res.ok) throw responseError(res, body);
      if (!isRecord(body) || typeof body.seq !== 'number' || typeof body.text !== 'string') {
        throw new ApiError('Invalid transcribe response', res.status);
      }
      return { seq: body.seq, text: body.text };
    },

    async summarize(input) {
      const { res, body } = await postAi('/ai/summary', input);
      if (!isRecord(body) || typeof body.summary !== 'string') {
        throw new ApiError('Invalid summary response', res.status);
      }
      return { summary: body.summary };
    },

    async todos(input) {
      const { res, body } = await postAi('/ai/todos', input);
      const items = isRecord(body) ? body.items : undefined;
      if (!Array.isArray(items) || !items.every((item): item is string => typeof item === 'string')) {
        throw new ApiError('Invalid todos response', res.status);
      }
      return { items };
    },
  };
}

/** The ApiError for a non-2xx response, with the server's error code and Retry-After. */
function responseError(res: Response, body: unknown): ApiError {
  const code = isRecord(body) && isApiErrorCode(body.error) ? body.error : undefined;
  return new ApiError(`Request failed with status ${res.status}`, res.status, {
    code,
    retryAfterMs: parseRetryAfter(res.headers.get('Retry-After')),
  });
}

/** `Retry-After` in ms: delta-seconds or an HTTP date; undefined when missing or unreadable. */
export function parseRetryAfter(value: string | null, now: number = Date.now()): number | undefined {
  if (value === null) return undefined;
  const trimmed = value.trim();
  if (/^\d+$/.test(trimmed)) return Number(trimmed) * 1000;
  const date = Date.parse(trimmed);
  return Number.isNaN(date) ? undefined : Math.max(0, date - now);
}

async function readJson(res: Response): Promise<unknown> {
  try {
    return await res.json();
  } catch {
    // Proxies and gateways may answer with HTML or plain text.
    return undefined;
  }
}

function isRecord(value: unknown): value is Record<string, unknown> {
  return typeof value === 'object' && value !== null;
}

 * sent on its own). Lists keep their markers, transcript gaps read [未轉寫片段].

export function aiNoteText(note: Note): string {

}

export interface AiAvailability {
  /** A recording runs, is stopping or is waiting for the microphone. */
  recording: boolean;
  /**
   * Backend URL and access key are set (isConfigured). A backend without a Groq key is not known
   * here: it answers 503 ai_unavailable, which the panel reports (AI_MESSAGES.unavailable).
   */
  configured: boolean;
  /** aiNoteText of the open note. */
  text: string;
}

/** The reason the AI buttons are disabled, or undefined when they can be used. */
export function aiDisabledReason({ recording, configured, text }: AiAvailability): string | undefined {
  if (recording) return AI_DISABLED_REASONS.recording;
  if (!configured) return AI_DISABLED_REASONS.notConfigured;
  // Gap markers alone are no content to summarise.
  if (text.replaceAll(TRANSCRIPT_GAP_TEXT, '').trim() === '') return AI_DISABLED_REASONS.empty;
  return undefined;
}

/** The panel message for a failed summarize() / todos() call. */
export function aiErrorMessage(error: unknown): string {
  if (!(error instanceof ApiError)) return AI_MESSAGES.failed;
  if (error.code === 'ai_unavailable') return AI_MESSAGES.unavailable;
  switch (error.status) {
    case 0:
      return AI_MESSAGES.network;
    case 400:
      return AI_MESSAGES.badRequest;
    case 401:
      return AI_MESSAGES.unauthorized;
    case 413:
      return AI_MESSAGES.tooLong;
    case 429:
      return AI_MESSAGES.rateLimited;
    default:
      return AI_MESSAGES.failed;
  }
}

  acknowledge(): void;

import type { JSONContent } from '@oc/shared';
import type { Editor } from '@tiptap/react';

/** Heading placed above the inserted task list. */
export const TODO_HEADING = '待辦事項';

/** One row of the 待辦事項 panel: the item text and whether the user ticked it. */
export interface TodoItem {
  text: string;
  checked: boolean;
}

const paragraph = (text: string): JSONContent => ({ type: 'paragraph', content: [{ type: 'text', text }] });

/** The summary's non-blank lines, trimmed: one paragraph each in the panel and in the note. */
export function summaryLines(summary: string): string[] {
  return summary
    .split(/\r?\n/)
    .map((line) => line.trim())
    .filter((line) => line !== '');
}

/** Inserts the summary at the very start of the note (one undo step); false when nothing was inserted. */
export function insertSummaryAtStart(editor: Editor, summary: string): boolean {
  const lines = summaryLines(summary);
  if (lines.length === 0 || editor.isDestroyed) return false;
  return editor.chain().insertContentAt(0, lines.map(paragraph)).run();
}

/**
 * Inserts a 待辦事項 heading and a task list with the items at the end of the note (one undo step).
 * An empty paragraph at the end of the note is replaced rather than left above the heading.
 * Items without text are skipped; false when nothing was inserted.
 */
export function insertTodosAtEnd(editor: Editor, items: TodoItem[]): boolean {
  const kept = items
    .map(({ text, checked }) => ({ text: text.trim(), checked }))
    .filter(({ text }) => text !== '');
  if (kept.length === 0 || editor.isDestroyed) return false;
  const blocks: JSONContent[] = [
    { type: 'heading', attrs: { level: 2 }, content: [{ type: 'text', text: TODO_HEADING }] },
    {
      type: 'taskList',
      content: kept.map(({ text, checked }) => ({ type: 'taskItem', attrs: { checked }, content: [paragraph(text)] })),
    },
  ];
  const { doc } = editor.state;
  const end = doc.content.size;
  const last = doc.lastChild;
  let range: number | { from: number; to: number } = end;
  if (last && last.type.name === 'paragraph' && last.content.size === 0) {
    range = { from: end - last.nodeSize, to: end };
  }
  return editor.chain().insertContentAt(range, blocks).run();
}

import type { AiRequest } from '@oc/shared';
import { useCallback, useEffect, useRef, useState } from 'react';
import { ApiError, createApiClient, type ApiClient } from '../api/apiClient';
import type { Settings } from '../settings/settingsStore';
import { aiErrorMessage } from './aiMessages';
import { aiPrivacy, type AiPrivacyStore } from './aiPrivacy';

export type AiKind = 'summary' | 'todos';

/** What the AI panel shows. `requestId` changes for every request, so the panel can be keyed on it. */
export type AiPanelState = { requestId: number } & (
  | { kind: AiKind; status: 'loading' }
  | { kind: 'summary'; status: 'done'; summary: string }
  | { kind: 'todos'; status: 'done'; items: string[] }
  | { kind: AiKind; status: 'error'; message: string }
);

export type AiClient = Pick<ApiClient, 'summarize' | 'todos'>;

/** Everything useAiAssistant touches outside React; tests replace these. */
export interface AiAssistantDeps {
  createClient(settings: Settings): AiClient;
  privacy: AiPrivacyStore;
}

export const defaultAiAssistantDeps: AiAssistantDeps = {
  createClient: (settings) => createApiClient({ baseUrl: settings.apiBaseUrl, accessKey: settings.accessKey }),
  privacy: aiPrivacy,
};

export interface UseAiAssistantOptions {
  settings: Settings;
  /** Called when the backend rejects the access key (401), e.g. to open the settings. */
  onUnauthorized?(): void;
  deps?: AiAssistantDeps;
}

export interface UseAiAssistantResult {
  /** The open AI panel, or null when it is closed. */
  panel: AiPanelState | null;
  /** The privacy notice must be shown; the request waits for acceptPrivacy(). */
  privacyPending: boolean;
  /** Opens the panel for `kind` with the note's title and text (after the privacy notice the first time). */
  request(kind: AiKind, input: AiRequest): void;
  /** Sends the last request again. */
  retry(): void;
  /** 我知道了: remembers the notice and sends the waiting request. */
  acceptPrivacy(): void;
  /** Closes the notice without sending anything. */
  cancelPrivacy(): void;
  /** Closes the panel (and the notice); a response still on its way is ignored. Stable across renders. */
  close(): void;
}

interface AiCall {
  kind: AiKind;
  input: AiRequest;
}

/** State and requests behind AI 摘要 and 待辦事項. Only the latest request may update the panel. */
export function useAiAssistant({
  settings,
  onUnauthorized,
  deps = defaultAiAssistantDeps,
}: UseAiAssistantOptions): UseAiAssistantResult {
  const [panel, setPanel] = useState<AiPanelState | null>(null);
  const [pending, setPending] = useState<AiCall | null>(null);
  const requestId = useRef(0);
  const lastCall = useRef<AiCall | null>(null);

  // After unmount no response may set state.
  useEffect(
    () => () => {
      requestId.current += 1;
    },
    [],
  );

  const run = ({ kind, input }: AiCall) => {
    const id = ++requestId.current;
    lastCall.current = { kind, input };
    setPanel({ requestId: id, kind, status: 'loading' });
    const client = deps.createClient(settings);
    const result: Promise<AiPanelState> =
      kind === 'summary'
        ? client
            .summarize(input)
            .then(({ summary }): AiPanelState => ({ requestId: id, kind: 'summary', status: 'done', summary }))
        : client
            .todos(input)
            .then(({ items }): AiPanelState => ({ requestId: id, kind: 'todos', status: 'done', items }));
    result.then(
      (next) => {
        if (requestId.current === id) setPanel(next);
      },
      (error: unknown) => {
        if (requestId.current !== id) return;
        console.error('AI request failed', error);
        setPanel({ requestId: id, kind, status: 'error', message: aiErrorMessage(error) });
        if (error instanceof ApiError && error.status === 401) onUnauthorized?.();
      },
    );
  };

  const close = useCallback(() => {
    requestId.current += 1;
    setPanel(null);
    setPending(null);
  }, []);

  return {
    panel,
    privacyPending: pending !== null,
    request(kind, input) {
      if (deps.privacy.isAcknowledged()) run({ kind, input });
      else setPending({ kind, input });
    },
    retry() {
      if (lastCall.current) run(lastCall.current);
    },
    acceptPrivacy() {
      if (!pending) return;
      deps.privacy.acknowledge();
      setPending(null);
      run(pending);
    },
    cancelPrivacy() {
      setPending(null);
    },
    close,
  };
}
```

import { useEffect, useId, useRef, type KeyboardEvent } from 'react';

export const AI_PRIVACY_TEXT = {
  title: 'AI 功能說明',
  body: '使用 AI 摘要或待辦事項時，筆記內容會送到 Groq 處理。',
  hint: '按「我知道了」後，這則說明就不會再出現。',
  accept: '我知道了',
  cancel: '取消',
} as const;

export interface AiPrivacyNoticeProps {
  /** 我知道了: remember the notice and send the request. */
  onAccept(): void;
  /** 取消, Escape or a click outside: close without sending. */
  onCancel(): void;
}

/** Shown once, before the first AI request: the note goes to Groq. Render it only while it is open. */
export function AiPrivacyNotice({ onAccept, onCancel }: AiPrivacyNoticeProps) {
  const titleId = useId();
  const bodyId = useId();
  const dialogRef = useRef<HTMLDivElement>(null);
  const acceptRef = useRef<HTMLButtonElement>(null);

  useEffect(() => {
    const opener = document.activeElement instanceof HTMLElement ? document.activeElement : null;
    acceptRef.current?.focus();
    return () => {
      if (opener?.isConnected) opener.focus();
    };
  }, []);

  const onKeyDown = (event: KeyboardEvent<HTMLDivElement>) => {
    if (event.key === 'Escape') {
      event.stopPropagation();
      onCancel();
      return;
    }
    if (event.key !== 'Tab' || !dialogRef.current) return;
    const buttons = Array.from(dialogRef.current.querySelectorAll<HTMLButtonElement>('button'));
    const first = buttons[0];
    const last = buttons[buttons.length - 1];
    if (!first || !last) return;
    if (event.shiftKey && document.activeElement === first) {
      event.preventDefault();
      last.focus();
    } else if (!event.shiftKey && document.activeElement === last) {
      event.preventDefault();
      first.focus();
    }
  };

  return (
    <div
      className="oc-dialog-backdrop"
      onMouseDown={(event) => {
        if (event.target === event.currentTarget) onCancel();
      }}
    >
      <div
        ref={dialogRef}
        className="oc-dialog oc-ai-privacy"
        role="dialog"
        aria-modal="true"
        aria-labelledby={titleId}
        aria-describedby={bodyId}
        onKeyDown={onKeyDown}
      >
        <h2 id={titleId} className="oc-dialog-title">
          {AI_PRIVACY_TEXT.title}
        </h2>
        <p id={bodyId} className="oc-ai-privacy-body">
          {AI_PRIVACY_TEXT.body}
        </p>
        <p className="oc-dialog-hint">{AI_PRIVACY_TEXT.hint}</p>
        <div className="oc-dialog-actions">
          <button type="button" className="oc-action-btn" onClick={onCancel}>
            {AI_PRIVACY_TEXT.cancel}
          </button>
          <button ref={acceptRef} type="button" className="oc-primary-btn" onClick={onAccept}>
            {AI_PRIVACY_TEXT.accept}
          </button>
        </div>
      </div>
    </div>
  );
}
```

import { Copy, X } from 'lucide-react';
import { useEffect, useId, useRef, useState } from 'react';
import { summaryLines, type TodoItem } from '../ai/insertAi';
import type { AiKind, AiPanelState } from '../ai/useAiAssistant';

export const AI_PANEL_TITLES: Record<AiKind, string> = { summary: 'AI 摘要', todos: '待辦事項' };

export const AI_PANEL_TEXT = {
  loading: 'AI 處理中…',
  retry: '重試',
  insertSummary: '插入到筆記開頭',
  copy: '複製',
  copied: '已複製',
  copyFailed: '無法複製，請手動選取文字',
  emptySummary: 'AI 沒有產生摘要，請稍後再試。',
  insertTodos: '插入到筆記結尾',
  noTodos: '這篇筆記裡沒有找到待辦事項。',
  close: '關閉 AI 面板',
} as const;

const defaultCopyText = async (text: string) => {
  const clipboard = globalThis.navigator?.clipboard;
  if (!clipboard?.writeText) throw new Error('Clipboard API is not available');
  await clipboard.writeText(text);
};

export interface AiPanelProps {
  state: AiPanelState;
  onClose(): void;
  onRetry(): void;
  onInsertSummary(summary: string): void;
  onInsertTodos(items: TodoItem[]): void;
  /** Why inserting is not possible right now (e.g. while recording); shown as the buttons' title. */
  insertDisabledReason?: string;
  /** Hides the panel from focus and assistive technology, e.g. while the note drawer is open. */
  inert?: boolean;
  /** navigator.clipboard.writeText by default; injectable for tests. */
  copyText?: (text: string) => Promise<void>;
}

/** Side panel for AI 摘要 and 待辦事項. Key it on state.requestId so every request starts fresh. */
export function AiPanel({
  state,
  onClose,
  onRetry,
  onInsertSummary,
  onInsertTodos,
  insertDisabledReason,
  inert,
  copyText = defaultCopyText,
}: AiPanelProps) {
  const titleId = useId();
  const headingRef = useRef<HTMLHeadingElement>(null);

  useEffect(() => {
    headingRef.current?.focus();
  }, []);

  return (
    <aside
      className="oc-ai-panel"
      aria-labelledby={titleId}
      inert={inert}
      onKeyDown={(event) => {
        if (event.key !== 'Escape') return;
        event.stopPropagation();
        onClose();
      }}
    >
      <div className="oc-ai-panel-header">
        <h2 id={titleId} ref={headingRef} tabIndex={-1}>
          {AI_PANEL_TITLES[state.kind]}
        </h2>
        <button type="button" className="oc-icon-btn" aria-label={AI_PANEL_TEXT.close} onClick={onClose}>
          <X size={16} aria-hidden="true" />
        </button>
      </div>
      <div className="oc-ai-panel-body">
        {state.status === 'loading' && (
          <p className="oc-ai-panel-status" role="status">
            {AI_PANEL_TEXT.loading}
          </p>
        )}
        {state.status === 'error' && (
          <>
            <p className="oc-ai-panel-error" role="alert">
              {state.message}
            </p>
            <div className="oc-ai-panel-actions">
              <button type="button" className="oc-action-btn" onClick={onRetry}>
                {AI_PANEL_TEXT.retry}
              </button>
            </div>
          </>
        )}
        {state.status === 'done' && state.kind === 'summary' && (
          <SummaryResult
            summary={state.summary}
            onInsert={onInsertSummary}
            insertDisabledReason={insertDisabledReason}
            copyText={copyText}
          />
        )}
        {state.status === 'done' && state.kind === 'todos' && (
          <TodoResult items={state.items} onInsert={onInsertTodos} insertDisabledReason={insertDisabledReason} />
        )}
      </div>
    </aside>
  );
}

function SummaryResult({
  summary,
  onInsert,
  insertDisabledReason,
  copyText,
}: {
  summary: string;
  onInsert(summary: string): void;
  insertDisabledReason?: string;
  copyText(text: string): Promise<void>;
}) {
  const [copyMessage, setCopyMessage] = useState('');
  const lines = summaryLines(summary);
  const empty = lines.length === 0;

  const copy = () => {
    copyText(summary).then(
      () => setCopyMessage(AI_PANEL_TEXT.copied),
      (error: unknown) => {
        console.error('Failed to copy the summary', error);
        setCopyMessage(AI_PANEL_TEXT.copyFailed);
      },
    );
  };

  return (
    <>
      {empty ? (
        <p className="oc-ai-panel-status">{AI_PANEL_TEXT.emptySummary}</p>
      ) : (
        <div className="oc-ai-summary">
          {lines.map((line, index) => (
            <p key={index}>{line}</p>
          ))}
        </div>
      )}
      <div className="oc-ai-panel-actions">
        <button
          type="button"
          className="oc-primary-btn"
          disabled={empty || insertDisabledReason !== undefined}
          title={insertDisabledReason}
          onClick={() => onInsert(summary)}
        >
          {AI_PANEL_TEXT.insertSummary}
        </button>
        <button type="button" className="oc-action-btn" disabled={empty} onClick={copy}>
          <Copy size={14} aria-hidden="true" />
          <span>{AI_PANEL_TEXT.copy}</span>
        </button>
      </div>
      <p className="oc-ai-panel-note" aria-live="polite">
        {copyMessage}
      </p>
    </>
  );
}

function TodoResult({
  items,
  onInsert,
  insertDisabledReason,
}: {
  items: string[];
  onInsert(items: TodoItem[]): void;
  insertDisabledReason?: string;
}) {
  const texts = items.map((item) => item.trim()).filter((item) => item !== '');
  const [checked, setChecked] = useState<boolean[]>(() => texts.map(() => false));

  if (texts.length === 0) return <p className="oc-ai-panel-status">{AI_PANEL_TEXT.noTodos}</p>;

  return (
    <>
      <ul className="oc-ai-todos">
        {texts.map((text, index) => (
          <li key={index}>
            <label>
              <input
                type="checkbox"
                checked={checked[index] ?? false}
                onChange={(event) => {
                  const value = event.target.checked;
                  setChecked((previous) => previous.map((old, i) => (i === index ? value : old)));
                }}
              />
              <span>{text}</span>
            </label>
          </li>
        ))}
      </ul>
      <div className="oc-ai-panel-actions">
        <button
          type="button"
          className="oc-primary-btn"
          disabled={insertDisabledReason !== undefined}
          title={insertDisabledReason}
          onClick={() => onInsert(texts.map((text, index) => ({ text, checked: checked[index] ?? false })))}
        >
          {AI_PANEL_TEXT.insertTodos}
        </button>
      </div>
    </>
  );
}
